   TAREFA A: Inserir hóspedes aleatórios 
   ========================================================================== */
CREATE OR REPLACE PROCEDURE prc_inserir_hospedes ( 
   p_qtd_hospedes IN NUMBER, 
   p_idade_min    IN NUMBER, 
   p_idade_max    IN NUMBER 
) IS 
   v_nome           VARCHAR2(100); 
   v_cidade         VARCHAR2(50); 
   v_idade          NUMBER; 
   v_data_nasc      DATE; 
   v_nome_aux       VARCHAR2(50); 
   v_sobrenome_aux  VARCHAR2(50); 
BEGIN 
   -- Validações (Tratamento de Erros - Competência C3)
   IF p_qtd_hospedes IS NULL OR p_qtd_hospedes <= 0 THEN 
      RAISE_APPLICATION_ERROR(-20001, 'A quantidade de hóspedes deve ser maior que zero.'); 
   END IF; 
 
   IF p_idade_min IS NULL OR p_idade_max IS NULL THEN 
      RAISE_APPLICATION_ERROR(-20002, 'Idade mínima e máxima são obrigatórias.'); 
   END IF; 
 
   IF p_idade_min <= 18 THEN 
      RAISE_APPLICATION_ERROR(-20003, 'A idade mínima deve ser maior que 18 anos.'); 
   END IF; 
 
   IF p_idade_max >= 65 THEN 
      RAISE_APPLICATION_ERROR(-20004, 'A idade máxima deve ser menor que 65 anos.'); 
   END IF; 
 
   IF p_idade_min > p_idade_max THEN 
      RAISE_APPLICATION_ERROR(-20005, 'A idade mínima não pode ser maior que a máxima.'); 
   END IF; 
 
   -- Loop de Inserção (Estruturas de Repetição - Competência C4)
   FOR i IN 1 .. p_qtd_hospedes LOOP 
      SELECT nome INTO v_nome_aux 
        FROM (SELECT nome FROM AUX_NOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
 
      SELECT sobrenome INTO v_sobrenome_aux 
        FROM (SELECT sobrenome FROM AUX_SOBRENOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
 
      v_nome := v_nome_aux || ' ' || v_sobrenome_aux; 
 
      SELECT cidade INTO v_cidade 
        FROM (SELECT cidade FROM AUX_CIDADE ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
 
      -- Sorteia idade dentro do intervalo e calcula data de nascimento
      v_idade := TRUNC(DBMS_RANDOM.VALUE(p_idade_min, p_idade_max + 1)); 
      v_data_nasc := TRUNC(ADD_MONTHS(SYSDATE, -v_idade * 12)); 
 
      INSERT INTO HOSPEDE (nome, cidade, dataNascimento) 
      VALUES (v_nome, v_cidade, v_data_nasc); 
   END LOOP; 
 
   COMMIT; 
END prc_inserir_hospedes; 
/

/* ==========================================================================
   TAREFA B: Inserir atendentes 
   ========================================================================== */
CREATE OR REPLACE PROCEDURE prc_inserir_atendentes ( 
   p_qtd_atendentes IN NUMBER 
) IS 
   v_nome         VARCHAR2(100); 
   v_nome_aux     VARCHAR2(50); 
   v_sobrenome    VARCHAR2(50); 
   v_id_superior  NUMBER; 
BEGIN 
   IF p_qtd_atendentes IS NULL OR p_qtd_atendentes <= 0 THEN 
      RAISE_APPLICATION_ERROR(-20010, 'A quantidade de atendentes deve ser maior que zero.'); 
   END IF; 

   -- 1º ATENDENTE: Ele é o superior de todos. O campo codSuperior será nulo (NULL)
   SELECT nome INTO v_nome_aux FROM (SELECT nome FROM AUX_NOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
   SELECT sobrenome INTO v_sobrenome FROM (SELECT sobrenome FROM AUX_SOBRENOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
   v_nome := v_nome_aux || ' ' || v_sobrenome; 

   -- Usando cláusula RETURNING para capturar o ID real gerado pelo banco
   INSERT INTO ATENDENTE (codSuperior, nome) 
   VALUES (NULL, v_nome)
   RETURNING codAtendente INTO v_id_superior; 

   -- DEMAIS ATENDENTES: Vinculados ao ID gerado no primeiro registro
   IF p_qtd_atendentes > 1 THEN
       FOR i IN 2 .. p_qtd_atendentes LOOP 
          SELECT nome INTO v_nome_aux FROM (SELECT nome FROM AUX_NOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
          SELECT sobrenome INTO v_sobrenome FROM (SELECT sobrenome FROM AUX_SOBRENOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1; 
          v_nome := v_nome_aux || ' ' || v_sobrenome; 
     
          INSERT INTO ATENDENTE (codSuperior, nome) 
          VALUES (v_id_superior, v_nome); 
       END LOOP; 
   END IF;

   COMMIT; 
END prc_inserir_atendentes; 
/

/* ==========================================================================
   TAREFA C: Inserir hospedagens (Completa e com Regras de Negócio)
   ========================================================================== */
CREATE OR REPLACE PROCEDURE prc_inserir_hospedagens ( 
   p_qtd_hospedagens IN NUMBER, 
   p_data_inicio     IN DATE, 
   p_data_fim        IN DATE 
) IS 
   v_cod_hospede      NUMBER; 
   v_cod_atendente    NUMBER; 
   v_data_entrada     DATE; 
   v_data_saida       DATE; 
   v_num_quarto       NUMBER; 
   v_valor_diaria     NUMBER(9,2); 
   v_dias_estadia     NUMBER; 
   v_dias_intervalo   NUMBER; 
   v_total_hospedes   NUMBER; 
   v_total_atendentes NUMBER; 
   
   -- Uso de Cursores dinâmicos para sorteio (Competência C5)
   CURSOR c_hospedes IS SELECT codHospede FROM (SELECT codHospede FROM HOSPEDE ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;
   CURSOR c_atendentes IS SELECT codAtendente FROM (SELECT codAtendente FROM ATENDENTE ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;
BEGIN 
   -- Validações básicas
   IF p_data_inicio >= p_data_fim THEN
      RAISE_APPLICATION_ERROR(-20020, 'A data de início deve ser menor que a data final.');
   END IF;

   -- Valida se existem hóspedes e atendentes cadastrados antes de rodar
   SELECT COUNT(*) INTO v_total_hospedes FROM HOSPEDE;
   SELECT COUNT(*) INTO v_total_atendentes FROM ATENDENTE;
   
   IF v_total_hospedes = 0 OR v_total_atendentes = 0 THEN
      RAISE_APPLICATION_ERROR(-20021, 'É preciso ter hóspedes e atendentes cadastrados previamente.');
   END IF;

   -- Descobre a diferença de dias no intervalo fornecido
   v_dias_intervalo := TRUNC(p_data_fim) - TRUNC(p_data_inicio);

   FOR i IN 1 .. p_qtd_hospedagens LOOP 
      -- Sorteia Hóspede abrindo o cursor
      OPEN c_hospedes; FETCH c_hospedes INTO v_cod_hospede; CLOSE c_hospedes;
      
      -- Sorteia Atendente abrindo o cursor
      OPEN c_atendentes; FETCH c_atendentes INTO v_cod_atendente; CLOSE c_atendentes;

      -- Sorteia um quarto entre 1 e 100
      v_num_quarto := TRUNC(DBMS_RANDOM.VALUE(1, 101));
      
      -- Sorteia o número de diárias (máximo 3 dias conforme enunciado)
      v_dias_estadia := TRUNC(DBMS_RANDOM.VALUE(1, 4)); 
      
      -- Sorteia um valor aleatório para a diária
      v_valor_diaria := ROUND(DBMS_RANDOM.VALUE(120, 450), 2);

      -- Regra do Quarto: Buscar a data de saída da hospedagem mais recente DESTE quarto
      -- Se houver registro, a nova hospedagem começa logo após a saída anterior
      BEGIN
         SELECT TRUNC(MAX(dataSaida))
           INTO v_data_entrada
           FROM HOSPEDAGEM
          WHERE numQuarto = v_num_quarto;
      EXCEPTION
         WHEN NO_DATA_FOUND THEN v_data_entrada := NULL;
      END;

      -- Se o quarto nunca foi usado ou a última saída foi antes do início global, sorteia uma data inicial válida
      IF v_data_entrada IS NULL OR v_data_entrada < p_data_inicio THEN
         v_data_entrada := p_data_inicio + TRUNC(DBMS_RANDOM.VALUE(0, v_dias_intervalo));
      ELSE
         -- Se já foi usado, a nova entrada é no mesmo dia da última saída (ou no dia seguinte)
         v_data_entrada := v_data_entrada + 1;
      END IF;

      v_data_saida := v_data_entrada + v_dias_estadia;

      -- Só realiza o insert se a estadia calculada couber dentro da data estipulada
      IF v_data_saida <= p_data_fim THEN
         INSERT INTO HOSPEDAGEM (codAtendente, codHospede, dataEntrada, dataSaida, numQuarto, valorDiaria) 
         VALUES (v_cod_atendente, v_cod_hospede, v_data_entrada, v_data_saida, v_num_quarto, v_valor_diaria); 
      END IF;
   END LOOP; 
 
   COMMIT; 
END prc_inserir_hospedagens;

-- TAREFA A: Inserir Hóspedes
CREATE OR REPLACE PROCEDURE prc_inserir_hospedes (
   p_qtd_hospedes IN NUMBER,
   p_idade_min    IN NUMBER,
   p_idade_max    IN NUMBER
) IS
   v_nome_aux     VARCHAR2(50);
   v_sobrenome_aux VARCHAR2(50);
   v_nome         VARCHAR2(100);
   v_cidade       VARCHAR2(50);
   v_idade        NUMBER;
   v_data_nasc    DATE;
BEGIN
   -- Validações (C3 - Tratamento de Erros)
   IF p_qtd_hospedes IS NULL OR p_qtd_hospedes <= 0 THEN
      RAISE_APPLICATION_ERROR(-20001, 'Quantidade de hóspedes deve ser maior que zero.');
   END IF;
   IF p_idade_min IS NULL OR p_idade_max IS NULL OR p_idade_min > p_idade_max THEN
      RAISE_APPLICATION_ERROR(-20002, 'Idade mínima e máxima inválidas.');
   END IF;
   IF p_idade_min < 18 THEN
      RAISE_APPLICATION_ERROR(-20003, 'Idade mínima deve ser maior ou igual a 18.');
   END IF;
   IF p_idade_max > 65 THEN
      RAISE_APPLICATION_ERROR(-20004, 'Idade máxima deve ser menor ou igual a 65.');
   END IF;

   FOR i IN 1..p_qtd_hospedes LOOP
      -- Sorteio aleatório
      SELECT nome INTO v_nome_aux 
      FROM (SELECT nome FROM AUX_NOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;

      SELECT sobrenome INTO v_sobrenome_aux 
      FROM (SELECT sobrenome FROM AUX_SOBRENOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;

      SELECT cidade INTO v_cidade 
      FROM (SELECT cidade FROM AUX_CIDADE ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;

      v_nome  := v_nome_aux || ' ' || v_sobrenome_aux;
      v_idade := TRUNC(DBMS_RANDOM.VALUE(p_idade_min, p_idade_max + 1));
      v_data_nasc := ADD_MONTHS(TRUNC(SYSDATE), -v_idade * 12);

      INSERT INTO HOSPEDE (nome, cidade, dataNascimento)
      VALUES (v_nome, v_cidade, v_data_nasc);
   END LOOP;

   COMMIT;
   DBMS_OUTPUT.PUT_LINE(p_qtd_hospedes || ' hóspedes inseridos com sucesso.');
EXCEPTION
   WHEN OTHERS THEN
      ROLLBACK;
      DBMS_OUTPUT.PUT_LINE('Erro na prc_inserir_hospedes: ' || SQLERRM);
      RAISE;
END prc_inserir_hospedes;

-- TAREFA B: Inserir Atendentes (VERSÃO CORRIGIDA)
CREATE OR REPLACE PROCEDURE prc_inserir_atendentes (
   p_qtd_atendentes IN NUMBER
) IS
   v_nome_aux     VARCHAR2(50);
   v_sobrenome    VARCHAR2(50);
   v_nome         VARCHAR2(100);
   v_cod_chefe    NUMBER;
BEGIN
   IF p_qtd_atendentes IS NULL OR p_qtd_atendentes < 1 THEN
      RAISE_APPLICATION_ERROR(-20010, 'Quantidade de atendentes deve ser maior que zero.');
   END IF;

   -- Cria o Chefe (Atendente 1)
   SELECT nome INTO v_nome_aux FROM (SELECT nome FROM AUX_NOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;
   SELECT sobrenome INTO v_sobrenome FROM (SELECT sobrenome FROM AUX_SOBRENOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;
   v_nome := v_nome_aux || ' ' || v_sobrenome;

   INSERT INTO ATENDENTE (codSuperior, nome) VALUES (NULL, v_nome);
   v_cod_chefe := SQL%ROWCOUNT; -- Pega o último codAtendente gerado

   -- Insere os demais atendentes com o chefe como superior
   FOR i IN 2..p_qtd_atendentes LOOP
      SELECT nome INTO v_nome_aux FROM (SELECT nome FROM AUX_NOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;
      SELECT sobrenome INTO v_sobrenome FROM (SELECT sobrenome FROM AUX_SOBRENOME ORDER BY DBMS_RANDOM.VALUE) WHERE ROWNUM = 1;
      v_nome := v_nome_aux || ' ' || v_sobrenome;

      INSERT INTO ATENDENTE (codSuperior, nome) VALUES (v_cod_chefe, v_nome);
   END LOOP;

   COMMIT;
   DBMS_OUTPUT.PUT_LINE(p_qtd_atendentes || ' atendentes inseridos com sucesso.');
EXCEPTION
   WHEN OTHERS THEN
      ROLLBACK;
      DBMS_OUTPUT.PUT_LINE('Erro na prc_inserir_atendentes: ' || SQLERRM);
      RAISE;
END prc_inserir_atendentes;
